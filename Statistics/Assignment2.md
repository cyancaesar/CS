$$
X \sim Binomial(10,0.1) 
$$
$$
n = 10,\hspace{5mm}p=0.1
$$
$$
P(X = x) = {n \choose x}p^{x}(1-p)^{n-x}
$$
$$
P(X = x) = {10 \choose x}0.1^{x}(0.9)^{10-x}
$$

$(i)\hspace{5mm}P(X < 2)$
$$
P(X < 2) = P(X=0)+P(X=1)=0.3486+0.3874=0.736
$$

$(ii)\hspace{5mm}P(2 < X < 5)$
$$
P(2<X<5) = P(X=3)+P(X=4)=0.0573+0.0111=0.0684
$$

$(iii)\hspace{5mm}E(X)$
$$
E(X)=\micro=np=10\times0.1=1
$$
$(iv)\hspace{5mm} E(X-2)$
$$
E(X-2)=E(X)-2=1-2=-1
$$
$(v)\hspace{5mm}var(x)$
$$
var(x)=\sigma^2=np(1-p)=1\times0.9=0.9
$$
